# Application dependencies for the Real-Time Crypto and Stock Data Platform.
#
# Pinned to exact versions so every machine (yours, CI, Docker) builds the same
# environment. Upgrade deliberately, one group at a time, and re-run tests.
#
# Apache Airflow is NOT in this file on purpose. It ships its own constraint
# files and pins hundreds of transitive packages, which would fight with the
# versions below. It lives in requirements-airflow.txt and, like in most
# production teams, runs in its own container/environment.

# --- Streaming: Kafka client -------------------------------------------------
# confluent-kafka wraps librdkafka (C), the same client used in production at
# scale. Much faster than the pure-Python kafka-python, and actively maintained.
# The [avro] extra adds the Schema Registry client and Avro serializer.
confluent-kafka[avro]==2.6.1
fastavro==1.13.1       # Avro encoder used by confluent-kafka's AvroSerializer

# --- Ingestion: talking to market data APIs ---------------------------------

tenacity==9.0.0        # retries with backoff when an API rate-limits or drops

# --- Data processing + lake storage -----------------------------------------

# --- Warehouse access --------------------------------------------------------

# --- Backend API + authentication -------------------------------------------
